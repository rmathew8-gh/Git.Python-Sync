#!/bin/bash
# Simplest script to list upstream and push-default for repos in ~/git-dir/RECENT

ROOT=~/git-dir/RECENT

find "$ROOT" -name .git -maxdepth 2 -type d -prune | while read -r gitdir; do
    repo=$(dirname "$gitdir")
    echo "Repo: ${repo#$ROOT/}"

    # Upstream: Try 'upstream' remote, then tracking remote
    u_url=$(git -C "$repo" remote get-url upstream 2>/dev/null)
    if [ -z "$u_url" ]; then
        u_remote=$(git -C "$repo" config "branch.$(git -C "$repo" rev-parse --abbrev-ref HEAD 2>/dev/null).remote")
        [ -n "$u_remote" ] && u_url=$(git -C "$repo" remote get-url "$u_remote" 2>/dev/null)
    fi

    # Push Default
    p_remote=$(git -C "$repo" config remote.pushDefault || git config --global remote.pushDefault)
    p_url=$(git -C "$repo" remote get-url "$p_remote" 2>/dev/null)

    echo "  Upstream: ${u_url:-N/A}"
    echo "  Push-Def: ${p_url:-N/A} (${p_remote:-none})"
    echo
done
#+end_src

#!/bin/bash
# Script to set upstream to push-default URL (if upstream is missing) and delete push-default config.

ROOT=~/git-dir/RECENT

find "$ROOT" -name .git -maxdepth 2 -type d -prune | while read -r gitdir; do
    repo=$(dirname "$gitdir")
    
    # Get the local push-default remote name
    p_remote=$(git -C "$repo" config --local remote.pushDefault)
    
    # If push-default is not set locally, nothing to convert or delete (ignoring global)
    [ -z "$p_remote" ] && continue

    # If 'upstream' remote doesn't exist, create it from push-default
    if ! git -C "$repo" remote get-url upstream &>/dev/null; then
        p_url=$(git -C "$repo" remote get-url "$p_remote" 2>/dev/null)
        if [ -n "$p_url" ]; then
            git -C "$repo" remote add upstream "$p_url"
            echo "${repo#$ROOT/}: Set upstream from $p_remote ($p_url)"
        fi
    fi

    # Always delete the local push-default config
    git -C "$repo" config --unset remote.pushDefault
    echo "${repo#$ROOT/}: Deleted push-default ($p_remote)"
done
#+end_src
